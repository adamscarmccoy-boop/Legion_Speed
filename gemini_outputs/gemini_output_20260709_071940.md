### 3. The Brain Client: `legion_brain_cli.py`
This is your new "Command Center." It uses a rich terminal interface to show you exactly how the LangGraph agent is "thinking" (tool calls) before it gives you the final answer.

```python
import os
import sys
import json
import time
from datetime import datetime

# Attempt to use 'rich' for beautiful terminal output, fallback to standard print
try:
    from rich.console import Console
    from rich.panel import Panel
    from rich.live import Live
    from rich.table import Table
    from rich import print as rprint
    USE_RICH = True
except ImportError:
    USE_RICH = False

# Import the orchestrator from your existing file
try:
    from legion_langgraph_brain import run_workflow_validated
except ImportError:
    print("❌ Error: Could not find 'legion_langgraph_brain.py'. Ensure it is in this directory.")
    sys.exit(1)

def run_cli():
    if USE_RICH:
        console = Console()
    else:
        class MockConsole:
            def print(self, *args, **kwargs): print(*args)
            def rule(self, title): print(f"\n{'='*len(title)}\n{title}\n{'='*len(title)}\n")
        console = MockConsole()

    console.rule("[bold cyan]LEGION SOVEREIGN INTELLIGENCE CLI[/bold cyan]")
    console.print("[dim]Model: Phi-3/Gemma via Ollama | Protocol: LangGraph[/dim]\n")

    if not USE_RICH:
        print("Tip: Install 'rich' (pip install rich) for a better experience.\n")

    while True:
        try:
            user_input = input("\n[USER] ❯ ").strip()
            if user_input.lower() in ["exit", "quit", "q"]:
                console.print("[yellow]Terminating session... Goodbye.[/yellow]")
                break
            if not user_input:
                continue

            console.print(f"\n[bold magenta]Thinking...[/bold magenta]")
            start_time = time.time()

            # --- EXECUTE THE BRAIN ---
            # This calls your validated LangGraph workflow
            response = run_workflow_validated(user_input)
            elapsed = time.time() - start_time

            # --- FORMAT OUTPUT ---
            if USE_RICH:
                # 1. Display Tool Usage (The "Reasoning Trace")
                if response.tools_called:
                    tool_table = Table(title="Agent Reasoning Trace", show_header=True, header_style="bold magenta")
                    tool_table.add_column("Tool Name", style="cyan")
                    tool_table.add_column("Result Summary", style="green")
                    
                    for i, tool_name in enumerate(response.tools_called):
                        summary = "Success"
                        if i < len(response.tool_results_summary):
                            # Try to grab a snippet of the tool data
                            res_data = response.tool_results_summary[i].get("data", "")
                            summary = str(res_data)[:60] + "..." if len(str(res_data)) > 60 else str(res_data)
                        tool_table.add_row(tool_name, summary)
                    
                    console.print(tool_table)

                # 2. Display Final Answer
                console.print(Panel(
                    response.answer,
                    title=f"[bold green]FINAL RESPONSE[/bold green] ({elapsed:.2f}s)",
                    subtitle=f"Model: {response.model} | Messages: {response.raw_messages}",
                    expand=False
                ))
            else:
                # Standard Print Fallback
                print("\n" + "-"*30)
                print(f"TOOLS CALLED: {', '.join(response.tools_called) if response.tools_called else 'None'}")
                print(f"ANSWER: {response.answer}")
                print(f"TIME: {elapsed:.2f}s")
                print("-"*30)

        except KeyboardInterrupt:
            break
        except Exception as e:
            console.print(f"[bold red]CRITICAL ERROR:[/bold red] {str(e)}")
            # Optional: print traceback for debugging
            # import traceback; traceback.print_exc()

if __name__ == "__main__":
    run_cli()
```

### Why this works:
*   **The Import Fix:** The supervisor was crashing because it was trying to import classes that lived in a file you hadn't written yet. `legion_sonic_engine_actors.py` provides the structure.
*   **The Constructor Fix:** `DSPAlignmentActor` was failing because the supervisor was trying to create it with 0 arguments, but the class requires a `lance_db_path`. I updated the supervisor's `actor_specs` to pass the path.
*   **The CLI Client:** Instead of just getting a raw JSON string, the new CLI uses `run_workflow_validated` to parse your Pydantic `SovereignLegionResponse`. It turns the messy tool-call logs into a clean "Reasoning Trace" table, allowing you to see if the agent actually used `query_sonic_core` or `search_vibe_vectors` to find its answer.
