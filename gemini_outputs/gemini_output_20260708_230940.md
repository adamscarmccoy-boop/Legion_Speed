To bridge the gap between the **Sonic-to-Visual Bridge** (Tier 2) and the **Vision Pipeline** (Execution), we need a stateful orchestrator. 

This LangGraph implementation treats each stage of your guide—Image Gen, HUD Scanner, and Video Compiler—as a discrete "Node." It manages the transfer of data (the Prompt and the File Paths) and handles the subprocess execution within your specific virtual environments.

### 🧠 The Orchestrator: `legion_vision_orchestrator.py`

```python
import os
import subprocess
import ray
import time
from typing import TypedDict, Optional, Annotated
from pydantic import BaseModel, Field
from langgraph.graph import StateGraph, END

# ─── IMPORTING THE BRIDGE ACTOR ───
# (Assuming the SonicToVisualBridgeActor is in the same directory or installed)
from sonic_to_visual_bridge import SonicToVisualBridgeActor, BridgePrompt

# ─── PATH CONFIGURATION (From your Guide) ───
VENV_GPU = r"E:\WEB CASE STUDY\.venv\Scripts\python.exe"
VENV_CPU = r"C:\WEB CASE STUDY\.venv\Scripts\python.exe"

SCRIPT_CLIP = "clip_generative_art.py"
SCRIPT_VISION = "ray_vision_pipeline.py"
SCRIPT_VIDEO = "feedback_visualizer.py"

# ─── STATE DEFINITION ───

class AgentState(TypedDict):
    """The 'Spine' of the orchestration session."""
    audio_path: str
    dna_vector: Optional[list[float]]
    visual_prompt: Optional[str]
    generated_art_path: Optional[str]
    hud_art_path: Optional[str]
    final_video_path: Optional[str]
    error: Optional[str]
    status: str  # "idle", "processing", "completed", "failed"

# ─── NODES: THE WORKFLOW STEPS ───

class LegionOrchestrator:
    def __init__(self):
        # Initialize Ray and the Bridge Actor
        if not ray.is_initialized():
            ray.init(ignore_reinit_error=True)
        self.bridge = SonicToVisualBridgeActor.remote()

    def extract_dna_node(self, state: AgentState) -> AgentState:
        """Step 1: Simulate/Call Sonic DNA Engine to get the 64-D vector."""
        print(f"🧬 [Node: DNA Extraction] Analyzing: {state['audio_path']}")
        try:
            # In production, replace this with: 
            # dna = SonicDNAEngine.extract(state['audio_path'])
            import numpy as np
            mock_dna = np.random.uniform(-1, 1, 64).tolist()
            return {**state, "dna_vector": mock_dna, "status": "analyzed"}
        except Exception as e:
            return {**state, "error": f"DNA Extraction Failed: {str(e)}", "status": "failed"}

    def semantic_bridge_node(self, state: AgentState) -> AgentState:
        """Step 2: Use the Ray Bridge Actor to turn Math into a CLIP Prompt."""
        print(f"🌉 [Node: Semantic Bridge] Projecting DNA to Language...")
        try:
            import numpy as np
            dna_np = np.array(state['dna_vector'])
            
            # Call the Ray Actor
            prompt_payload = ray.get(self.bridge.bridge_dna_to_prompt.remote(dna_np))
            
            print(f"✨ [Prompt Generated]: {prompt_payload.llm_expanded_prompt[:60]}...")
            return {**state, "visual_prompt": prompt_payload.llm_expanded_prompt, "status": "prompted"}
        except Exception as e:
            return {**state, "error": f"Bridge Failed: {str(e)}", "status": "failed"}

    def image_generation_node(self, state: AgentState) -> AgentState:
        """Step 3: Execute the GPU-bound CLIP optimization."""
        print(f"🎨 [Node: Image Gen] Running CLIP Optimization on GPU...")
        try:
            # We pass the prompt via Environment Variable so the script can read it
            # This avoids complex CLI argument parsing for long text prompts
            env = os.environ.copy()
            env["CLIP_PROMPT"] = state['visual_prompt']
            
            # Run the script using the GPU VENV
            result = subprocess.run(
                [VENV_GPU, SCRIPT_CLIP],
                env=env,
                capture_output=True,
                text=True,
                check=True
            )
            print(f"✅ CLIP Output: {result.stdout.strip()}")

            # Logic to find the file created (based on your guide's naming pattern)
            import glob
            import os
            outputs = glob.glob(r"C:\WEB CASE STUDY\mastered_output\generated_art_*.png")
            latest_art = max(outputs, key=os.path.getctime) if outputs else None
            
            if not latest_art:
                raise Exception("CLIP finished but no output file found.")

            return {**state, "generated_art_path": latest_art, "status": "image_ready"}
        except subprocess.CalledProcessError as e:
            return {**state, "error": f"CLIP GPU Error: {e.stderr}", "status": "failed"}
        except Exception as e:
            return {**state, "error": f"Image Gen Failed: {str(e)}", "status": "failed"}

    def vision_scanner_node(self, state: AgentState) -> AgentState:
        """Step 4: Run the CPU-bound HUD Scanner."""
        print(f"🤖 [Node: Vision Scanner] Running Cyber HUD Overlay...")
        try:
            # Run the script using the CPU VENV
            result = subprocess.run(
                [VENV_CPU, SCRIPT_VISION],
                capture_output=True,
                text=True,
                check=True
            )
            print(f"✅ Vision Output: {result.stdout.strip()}")

            hud_path = r"C:\WEB CASE STUDY\mastered_output\artwork_creation\cyber_hud_art.png"
            return {**state, "hud_art_path": hud_path, "status": "vision_ready"}
        except subprocess.CalledProcessError as e:
            return {**state, "error": f"Vision CPU Error: {e.stderr}", "status": "failed"}
        except Exception as e:
            return {**state, "error": f"Vision Failed: {str(e)}", "status": "failed"}

    def video_compiler_node(self, state: AgentState) -> AgentState:
        """Step 5: Compile the final reactive social reel."""
        print(f"🔊 [Node: Video Compiler] Muxing Audio and Visuals...")
        try:
            result = subprocess.run(
                [VENV_CPU, SCRIPT_VIDEO],
                capture_output=True,
                text=True,
                check=True
            )
            print(f"✅ Video Output: {result.stdout.strip()}")

            final_path = r"C:\WEB CASE STUDY\mastered_output\feedback_test\feedback_reel_final.mp4"
            return {**state, "final_video_path": final_path, "status": "completed"}
        except subprocess.CalledProcessError as e:
            return {**state, "error": f"Video Error: {e.stderr}", "status": "failed"}
        except Exception as e:
            return {**state, "error": f"Video Failed: {str(e)}", "status": "failed"}

# ─── GRAPH CONSTRUCTION ───

def create_orchestration_graph():
    orch = LegionOrchestrator()
    workflow = StateGraph(AgentState)

    # Add Nodes
    workflow.add_node("extract_dna", orch.extract_dna_node)
    workflow.add_node("semantic_bridge", orch.semantic_bridge_node)
    workflow.add_node("image_gen", orch.image_generation_node)
    workflow.add_node("vision_scanner", orch.vision_scanner_node)
    workflow.add_node("video_compiler", orch.video_compiler_node)

    # Define Edges (The Sequence)
    workflow.set_entry_point("extract_dna")
    workflow.add_edge("extract_dna", "semantic_bridge")
    workflow.add_edge("semantic_bridge", "image_gen")
    workflow.add_edge("image_gen", "vision_scanner")
    workflow.add_edge("vision_scanner", "video_compiler")
    workflow.add_edge("video_compiler", END)

    # Error Handling (Conditional Edges)
    # If any node sets status to 'failed', we route to END
    def check_for_errors(state: AgentState):
        if state.get("status") == "failed":
            return "end"
        return "continue"

    workflow.add_conditional_edges(
        "extract_dna", check_for_errors, {"end": END, "continue": "semantic_bridge"}
    )
    # (In a production version, you would add these checks after every node)

    return workflow.compile()

# ─── EXECUTION ENTRY POINT ───

if __name__ == "__main__":
    print("🚀 LEGION VISION ORCHESTRATOR STARTING...")
    
    # Initialize the Graph
    app = create_orchestration_graph()

    # Define the initial session
    initial_state = {
        "audio_path": r"E:\DJSUSAN\LEGION\what a waste-2 (Edit).wav",
        "dna_vector": None,
        "visual_prompt": None,
        "generated_art_path": None,
        "hud_art_path": None,
        "final_video_path": None,
        "error": None,
        "status": "idle"
    }

    # Run the Pipeline
    final_output = app.invoke(initial_state)

    # Final Summary
    print("\n" + "="*30)
    print("🏁 SESSION SUMMARY")
    print("="*30)
    if final_output["status"] == "completed":
        print(f"✅ SUCCESS!")
        print(f"🎬 Final Reel: {final_output['final_video_path']}")
        print(f"📝 Prompt Used: {final_output['visual_prompt']}")
    else:
        print(f"❌ FAILED")
        print(f"🚨 Error: {final_output['error']}")
    print("="*30)
```

### 📐 Why this works for your architecture:

1.  **Hardware Awareness:** The Orchestrator knows exactly which script goes to the **E: drive (GPU)** and which scripts go to the **C: drive (CPU)** by mapping them to your specific `.venv` paths.
2.  **The "Brain" Connection:** It uses `ray.get()` to pull the intelligence from the `SonicToVisualBridgeActor`, turning a math vector into a text string.
3.  **Subprocess Isolation:** By using `subprocess.run`, we prevent the Python memory of the Orchestrator from being bloated by the heavy weights of CLIP or Faster R-CNN. Each script runs, cleans up its GPU/CPU memory, and returns control to the graph.
4.  **Stateful Traceability:** If the pipeline fails at the "Vision Scanner" stage, the `AgentState` will tell you exactly where it died (e.g., `status: vision_ready` was never reached), making debugging your local system much easier.
