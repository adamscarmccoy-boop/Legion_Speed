# outreach_schemas.py
from __future__ import annotations
from pydantic import BaseModel, HttpUrl, Field
from pathlib import Path
from datetime import datetime
from typing import List, Dict, Optional


class ProspectRecord(BaseModel):
    name: str
    slug: str
    website: HttpUrl
    contact_email: Optional[str] = None
    sample_pack_url: Optional[HttpUrl] = None


class SegmentAlignment(BaseModel):
    segment_name: str
    matched_baseline: str
    alignment_score: float = Field(..., ge=0.0, le=1.0)
    pass_threshold: bool


class ForestVerdict(BaseModel):
    top_style_match: str
    style_confidence: float = Field(..., ge=0.0, le=1.0)
    is_anomaly: bool
    marketing_score: float = Field(..., ge=0.0, le=1.0)


class PackReport(BaseModel):
    prospect: ProspectRecord
    pack_name: str
    pack_url: HttpUrl
    analyzed_at: datetime
    total_segments: int
    alignment_pass_rate: float = Field(..., ge=0.0, le=1.0)
    avg_alignment: float = Field(..., ge=0.0, le=1.0)
    best_segment_score: float = Field(..., ge=0.0, le=1.0)
    fire_test_duration_ms: float
    forest_verdict: ForestVerdict
    segment_breakdown: List[SegmentAlignment]
    category_distribution: Dict[str, int]
    key_findings: List[str]
    comparable_to: List[str]


class EmailDraft(BaseModel):
    subject: str
    body: str
    cta: str
    send_to: Optional[str] = None
    attachment_paths: List[Path] = []

# save_helpers.py
import json
import hashlib
from pathlib import Path
from datetime import datetime
from typing import Tuple
from outreach_schemas import PackReport, EmailDraft, ProspectRecord


OUTREACH_ROOT = Path("outreach")


def checksum_bytes(b: bytes) -> str:
    return hashlib.sha256(b).hexdigest()


def ensure_outreach_dir(slug: str) -> Path:
    out = OUTREACH_ROOT / slug
    out.mkdir(parents=True, exist_ok=True)
    return out


def save_pack_report(report: PackReport) -> Path:
    out = ensure_outreach_dir(report.prospect.slug)
    json_path = out / "pack_report.json"
    json_path.write_text(report.json(indent=2, ensure_ascii=False), encoding="utf-8")
    return json_path


def save_email_draft(email: EmailDraft, slug: str) -> Path:
    out = ensure_outreach_dir(slug)
    email_path = out / "email_draft.json"
    email_path.write_text(email.json(indent=2, ensure_ascii=False), encoding="utf-8")
    return email_path


def save_html_report(html_bytes: bytes, slug: str) -> Path:
    out = ensure_outreach_dir(slug)
    html_path = out / "report.html"
    html_path.write_bytes(html_bytes)
    return html_path


def write_index(slug: str, report_path: Path, html_path: Path, email_path: Path, status: str = "pending_review") -> Path:
    out = ensure_outreach_dir(slug)
    meta = {
        "slug": slug,
        "saved_at": datetime.utcnow().isoformat() + "Z",
        "pack_report": str(report_path.name),
        "report_html": str(html_path.name),
        "email_draft": str(email_path.name),
        "report_checksum": checksum_bytes(report_path.read_bytes()),
        "html_checksum": checksum_bytes(html_path.read_bytes()),
        "status": status
    }
    index_path = out / "index.json"
    index_path.write_text(json.dumps(meta, indent=2), encoding="utf-8")
    return index_path

def get_template_env(template_dir: str = "templates") -> Environment:
    if os.path.isdir(template_dir) and os.path.exists(os.path.join(template_dir, "report_template.html")):
        env = Environment(
            loader=FileSystemLoader(template_dir),
            autoescape=select_autoescape(["html", "xml"])
        )
        return env
    # fallback to Template from string
    env = Environment(autoescape=select_autoescape(["html", "xml"]))
    env.globals["default_template"] = Template(DEFAULT_TEMPLATE)
    return env


def render_report_html(report: PackReport, template_dir: str = "templates") -> bytes:
    env = get_template_env(template_dir)
    if os.path.isdir(template_dir) and os.path.exists(os.path.join(template_dir, "report_template.html")):
        template = env.get_template("report_template.html")
        html_out = template.render(report=report.dict())
    else:
        template = env.globals["default_template"]
        html_out = template.render(report=report.dict())
    return html_out.encode("utf-8")

# preflight.py
import subprocess
import sys
import os
import time


def kill_stale_ray_processes_windows():
    """
    Attempt to kill common stale Ray processes on Windows before starting a cluster.
    Uses taskkill to force-stop raylet and stale python processes older than 10 minutes.
    """
    try:
        # kill raylet if present
        subprocess.run(["taskkill", "/F", "/IM", "raylet.exe"], check=False, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        # kill any python processes that look stale (best-effort)
        # NOTE: This is conservative; adjust filters as needed.
        subprocess.run(["powershell", "-Command",
                        "Get-Process python -ErrorAction SilentlyContinue | Where-Object { $_.StartTime -lt (Get-Date).AddMinutes(-10) } | Stop-Process -Force"],
                       check=False, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
    except Exception:
        pass


def preflight_check():
    # platform-specific preflight
    if os.name == "nt":
        kill_stale_ray_processes_windows()
    # small sleep to allow ports to free
    time.sleep(0.5)

def build_pack_report(prospect: ProspectRecord, pack_path: Path, fire_result: dict) -> PackReport:
    segs = [SegmentAlignment(**s) for s in fire_result["segment_breakdown"]]
    forest = ForestVerdict(**score_alignment({}))
    report = PackReport(
        prospect=prospect,
        pack_name=pack_path.name,
        pack_url=prospect.sample_pack_url or prospect.website,
        analyzed_at=datetime.utcnow(),
        total_segments=fire_result["total_segments"],
        alignment_pass_rate=fire_result["alignment_pass_rate"],
        avg_alignment=fire_result["avg_alignment"],
        best_segment_score=fire_result["best_segment_score"],
        fire_test_duration_ms=fire_result["fire_test_duration_ms"],

        category_distribution={"drops": 9, "builds": 23},  # placeholder

            f"Avg alignment {(fire_result['avg_alignment']*100):.1f}%",
            f"{forest.top_style_match} match {forest.style_confidence:.2f}"
        ],
        comparable_to=[forest.top_style_match]
    )
    return report


def ensure_ray():
    if not ray.is_initialized():
        ray.init(ignore_reinit_error=True)


def main():
    parser = argparse.ArgumentParser(prog="weaponize.py")
    parser.add_argument("--target", required=True, help="Target domain or seed URL (e.g., splice.com)")
    parser.add_argument("--packs", type=int, default=1, help="Number of prospects/packs to process")
    parser.add_argument("--out", default="outreach", help="Output folder")
    args = parser.parse_args()

    # preflight
    preflight_check()

    # ensure output root
    os.environ["OUTREACH_ROOT"] = args.out

    # scrape prospects
    raw_prospects = scrape_prospects(args.target, limit=args.packs)
    prospects = []
    for p in raw_prospects:
        try:
            prospects.append(ProspectRecord(**p))
        except Exception:
            # skip invalid prospect records
            continue

    # initialize Ray (actors assumed to be used inside run_fire_test)
    try:
        ensure_ray()
    except Exception:
        pass

    for prospect in prospects:
        slug = prospect.slug
        staging_dir = Path(args.out) / slug / "staging"
        staging_dir.mkdir(parents=True, exist_ok=True)

        # download representative pack (first pack URL or fallback)
        pack_url = prospect.sample_pack_url or prospect.website
        pack_path = download_pack(pack_url, staging_dir)

        # run fire test (this may use Ray actors internally)
        fire_result = run_fire_test(str(pack_path))

        # build PackReport
        report = build_pack_report(prospect, Path(pack_path), fire_result)

        # render HTML
        html_bytes = render_report_html(report)

        # generate email draft
        email = generate_email_draft(report, to_email=prospect.contact_email)

        # save artifacts
        report_path = save_pack_report(report)
        html_path = save_html_report(html_bytes, slug)
        email_path = save_email_draft(email, slug)
        index_path = write_index(slug, report_path, html_path, email_path, status="pending_review")

        print(f"Saved outreach packet for {slug} -> {index_path}")

    # shutdown ray if we started it here
    try:
        if ray.is_initialized():
            ray.shutdown()
    except Exception:
        pass


if __name__ == "__main__":
    main()
