template<typename Float, int N>
struct FemSolver {
    // N = number of finite elements on the membrane
    Float K[N][N];    // stiffness matrix (precomputed per timbre)
    Float M[N][N];    // mass matrix (diagonalized)
    Float B[N];       // damping vector

struct alignas(64) AudioControlBlock {
    // Seqlock: readers spin on a parity counter, writers increment twice
    std::atomic<uint64_t> sequence;  // starts at 0
    
    // Vector model output — 16 floats, fits in 2 cache lines
    alignas(32) float params[16];
    
    static_assert(sizeof(decltype(params)) == 64);  // fits one half-cache-line
    
    // Reader (audio callback) — never blocks
    auto read_params() {
        uint64_t seq_before, seq_after;
        float local[16];
        do {
            seq_before = sequence.load(std::memory_order_acquire);
            __builtin_memcpy(local, params, sizeof(params));
            std::atomic_thread_fence(std::memory_order_acquire);
            seq_after = sequence.load(std::memory_order_acquire);
        } while (seq_before & 1 || seq_before != seq_after);
        // seq was even and unchanged → params are consistent
        return local;
    }
};
The audio thread does a simple spin-reading of the seqlock. No mutex, no syscall, no cache-line bouncing. The mmap region is pinned in physical memory with mlockall(MCL_FUTURE).3.     const float* __restrict__ params,  // 5 sets of 16 params each
    float* __restrict__ output         // stereo interleaved
) {
    // Each warp handles one voice
    int voice = blockIdx.x * 5 + (threadIdx.x >> 5);
    int lane  = threadIdx.x & 0x1f;
    
    // Registers for DSP state — no global memory access per sample
    float phase, envelope, filter_state, modulation;
    
    // Read params from shared memory (loaded once per block)
    __shared__ float block_params[5][16];
    if (lane < 16) block_params[voice][lane] = params[voice * 16 + lane];
    __syncwarp();
    
    auto p = block_params[voice];
    
    for (int sample = 0; sample < BLOCK_SIZE; ++sample) {
        // All DSP math in registers. Warp shuffle for voice synchronization.
        // 5 voices, 1 warp each, 32 registers used = full occupancy.
        
        phase += p[0];                    // frequency
        envelope = adsr(phase, p+1);       // 4 params
        float voice_out = osc(phase, p[5]) * envelope;
        
        voice_out = filter(voice_out, p[6], p[7], &filter_state);
        
        // Lane 0 = L, Lane 1 = R
        if (lane < 2) output[sample * 2 + lane] += voice_out * p[lane + 14];
    }
}
