def extract_real_audio_dna_local(audio_path: str) -> List[float]:
    """Extracts a 41-dim Sonic DNA vector from an audio file using librosa."""
    if not os.path.exists(audio_path):
        raise FileNotFoundError(f"Required audio file not found: {audio_path}")
    y, sr = librosa.load(audio_path, sr=22050, mono=True, duration=5) # Load first 5 seconds
    mfcc = librosa.feature.mfcc(y=y, sr=sr, n_mfcc=41)
    dna_vector = np.mean(mfcc, axis=1).astype(np.float32).tolist()
    if len(dna_vector) < 41:
        dna_vector = np.pad(dna_vector, (0, 41 - len(dna_vector))).tolist()
    return dna_vector[:41]
