def download_clip(query, output_name, duration="00:00:15"):

    # Use sys.executable -m yt_dlp for better compatibility with venvs

        "-o", output_name + ".m4a",
        f"ytsearch1:{query}"
    ]
    # For more verbose output during install, remove DEVNULL, but keeping for now.
    subprocess.run(cmd, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)

# 2. EXTRACT DSP FOOTPRINT
def extract_dsp_footprint(filepath):
    # Try m4a or wav
    audio_path = filepath + ".m4a"
    if not os.path.exists(audio_path):
        audio_path = filepath + ".wav" # Fallback if for some reason yt-dlp saves wav
        if not os.path.exists(audio_path):
            print(f"File {filepath}.m4a or {filepath}.wav not found!")
            # Return dummy values to prevent crashing, but indicate failure
            return 100.0, np.zeros(13), np.zeros(1024), 22050
    
    y, sr = librosa.load(audio_path, sr=22050)
    
    # Pitch
    # Increased fmin to avoid very low rumble, more robust for speech
    f0, voiced_flag, _ = librosa.pyin(y, fmin=librosa.note_to_hz('C3'), fmax=librosa.note_to_hz('C7'))
    valid_f0 = f0[voiced_flag]
    mean_pitch = np.median(valid_f0) if len(valid_f0) > 0 else 100.0
    
    # Formants / MFCCs
    mfccs = librosa.feature.mfcc(y=y, sr=sr, n_mfcc=13)
    mean_mfccs = np.mean(mfccs, axis=1)
    
    return mean_pitch, mean_mfccs, y, sr

# snoop_voice_app.py (modifications)
import pyaudio, numpy as np, torch, sys, os, threading, time
from pedalboard import Pedalboard, Compressor, PitchShift, LowpassFilter, Delay

running = False

def run_audio(ui_status_callback=None):
    global running
    try:
        base_dir = os.path.dirname(os.path.abspath(__file__))
        shift = float(torch.load(os.path.join(base_dir, "snoop_pitch_delta.pt"))[0])
        board = Pedalboard([Compressor(threshold_db=-15, ratio=4), PitchShift(semitones=shift), LowpassFilter(cutoff_frequency_hz=2500), Delay(delay_seconds=0.15, mix=0.1)])
        
        p = pyaudio.PyAudio()

        # --- NEW: Programmatically select BlackHole 2ch as output ---
        blackhole_index = -1
        # Loop through available devices to find "BlackHole 2ch"
        for i in range(p.get_device_count()):
            device_info = p.get_device_info_by_index(i)
            if "BlackHole 2ch" in device_info['name']: # Use "BlackHole 2ch" to be specific
                if device_info['maxOutputChannels'] > 0: # Ensure it's an output device
                    blackhole_index = i
                    break

        if blackhole_index == -1:
            error_msg = "Error: BlackHole 2ch virtual audio cable not found. Please ensure it's installed."
            print(error_msg)
            if ui_status_callback: ui_status_callback(f"Status: ERROR! {error_msg}")
            # Fallback to default output if BlackHole isn't found, but log the error
            stream_out = p.open(format=pyaudio.paFloat32, channels=1, rate=48000, output=True, frames_per_buffer=1024)
        else:
            stream_out = p.open(format=pyaudio.paFloat32, channels=1, rate=48000, output=True, frames_per_buffer=1024, output_device_index=blackhole_index)
        # --- END NEW ---

        # Input stream will still use the system's default input device, as per original instructions.
        stream_in = p.open(format=pyaudio.paFloat32, channels=1, rate=48000, input=True, frames_per_buffer=1024)
        
        if ui_status_callback: ui_status_callback("Status: LIVE 🎙️")
        while running:
            data = stream_in.read(1024, exception_on_overflow=False)
            audio = np.frombuffer(data, dtype=np.float32)
            out = board(audio, sample_rate=48000)
            stream_out.write(np.clip(out, -1.0, 1.0).astype(np.float32).tobytes())
    except Exception as e:
        print(f"Audio Error: {e}")
        if ui_status_callback: ui_status_callback(f"Status: ERROR! {e}")
    finally:
        try: stream_in.close(); stream_out.close(); p.terminate()
        except: pass
        if ui_status_callback: ui_status_callback("Status: OFF")
